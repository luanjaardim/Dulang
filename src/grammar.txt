# nl -> end of line (new line)

body ::== ( sttm )*

inner-body ::== "{" body "}"

sttm ::==
	cond
	var ASSIGN expr nl
	BACK expr? nl
	STOP expr? nl
	SKIP nl

cond ::==
	"if" comparison inner-body (elif | else)?
elif ::==
	"elif" comparison inner-body (elif | else)?
else ::==
	"else" inner-body
	
loop ::==
	"while" comparison inner-body
	"loop" inner-body

expr ::==
	bitwise
	func
	cond
	fn-call

func ::==
	"|" (var ( "," var )*)? "|" ( "->" type )? expr
	"|" (var ( "," var )*)? "|" ( "->" type )? inner-body

var ::==
	ID ("::" type)?

fn-call ::==
	ID "(" ")"   # Function with no parameters
	ID expr*

bitwise ::==
	BNOT bitwise
	comparison ((BAND | BOR | BXOR | SHL | SHR) comparison )*

comparison ::==
	NOT comparison
	arith ((AND | OR | GRT | GRE | LET | LEE | EQ | NEQ) arith)*

arith ::== factor ((ADD | SUB) factor)*

factor ::== unary ((MUL | DIV) unary)*

unary ::==
	primary
	(ADD | SUB) unary

primary ::==
	"(" expr ")"
	ID
	INTEGER
	REAL
	STRING
	CHAR

type ::==
	union_type ("->" union_type)*  # Function type

union_type ::==
	tuple_type ("^" tuple_type)*  # Union type

tuple_type ::==
	basic_types ("&" basic_types)*  # Tuple type

basic_types ::==
	"(" type ")"
	"i32"
	"u32"
	"char"
	"f32"
	"f64"
	"bool"
