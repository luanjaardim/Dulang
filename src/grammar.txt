# nl -> end of line (new line)


body ::== ( sttm )*

inner-body ::== "{" body "}"

sttm ::==
	cond
	var ASSIGN expr nl
	BACK expr nl
	BREAK expr nl
	SKIP

cond ::==
	"if" comparison inner-body (elif | else)?
elif ::==
	"elif" comparison inner-body (elif | else)?
else ::==
	"else" inner-body
	
expr ::==
	bitwise
	func
	cond

func ::==
	"|" (var)? ( "," var )* "|" ( "->" type )? expr
	"|" (var)? ( "," var )* "|" ( "->" type )? inner-body

var ::==
	ID ("::" type)?

bitwise ::==
	BNOT bitwise
	comparison ( (BAND | BOR | BXOR | SHL | SHR) comparison )*

comparison ::==
	NOT comparison
	arith ((GRT | GRE | LET | LEE | EQ | NEQ) arith)*

arith ::== factor ((ADD | SUB) factor)*

factor ::== unary ((MUL | DIV) unary)*

unary ::==
	primary
	(ADD | SUB) unary

primary ::==
	"(" expr ")"
	ID
	INTEGER
	REAL
	STRING
	CHAR

type ::==
	"i32"
	"u32"
	"char"
	"f32"
	"f64"
	"bool"
