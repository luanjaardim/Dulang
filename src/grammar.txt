# nl -> end of line (new line)

body ::== ( sttm )*

inner-body ::== "{" body "}"

sttm ::==
	module
	cond
	assign
	loop
	BACK expr? nl
	STOP expr? nl
	SKIP nl
	fn-call

module ::==
	ID ASSIGN "mod" (MOD_ACCESS | ID) nl
	ID ASSIGN "mod" inner-body

cond ::==
	"if" comparison inner-body (elif | else)?
elif ::==
	"elif" comparison inner-body (elif | else)?
else ::==
	"else" inner-body

assign ::==
	var ASSIGN expr nl               # Defining constants
	"var" ID ASSIGN "pass" nl        # Only inside Struct definitions
	
loop ::==
	"while" comparison inner-body
	"loop" inner-body

expr ::==
	parsed-expr ("::" type)?

parsed-expr :: ==
	struct
	array
	func
	type
	sub-expr

var ::==
	"var"? ID ("::" type)?

struct ::==
	"{" struct-body "}"

struct-body ::==
	struct-items (";" struct-items)*

struct-items ::==
	assign
	var    # TODO

array ::==
	"[" (expr ("," expr)*)? "]"

func ::==
	TODO: "(" (var ( "," var )*)? ")" type? ":" expr
	"(" (var ( "," var )*)? ")" type? ":" (inner-body | func)

sub-expr ::==
	fn-call
	unary
	bitwise

fn-call ::==
	id "none"   # Function with no parameters
	id expr*
	fn-call ("<<" fn-call)*
	fn-call (">>" fn-call)*

bitwise ::==
	BNOT bitwise
	comparison ((BAND | BOR | BXOR | SHL | SHR) comparison )*

comparison ::==
	NOT comparison
	indexing ((AND | OR | GRT | GRE | LET | LEE | EQ | NEQ) indexing)*

indexing ::==
	arith ("@" arith)?   # Dereference pointers with some offset

arith ::== factor ((ADD | SUB) factor)*

factor ::== primary ((MUL | DIV) primary)*

unary ::==
	"@" unary               # Dereference pointers
	"&" primary             # Reference as pointers
	"&var" primary          # Reference as variable pointers
	(ADD | SUB) primary
	primary

primary ::==
	"(" expr ("," expr)* ")"
	id
	INTEGER
	REAL
	STRING
	CHAR
	FALSE
	TRUE

id ::==
	id inner-body?
	ID
	STRUCT_ACCESS
	MOD_ACCESS

type ::==
	union_type ("->" union_type)*  # Function type

union_type ::==
	tuple_type ("|" tuple_type)*  # Union type

tuple_type ::==
	ptr_type ("^" ptr_type)*  # Tuple type

ptr_type ::==
	"&" basic_types           # Pointer to types
	"&var" basic_types        # Variable pointer to types
	basic_types

basic_types ::==
	"(" type ")"
	"i%d"
	"u%d"
	"f%d"
	"char"
	"bool"
	"type"
	"none"
	ID
