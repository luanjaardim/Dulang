# nl -> end of line (new line)


body ::== ( sttm )*
sttm ::==
	BACK expr nl
	BREAK expr nl
	SKIP
	ID ASSIGN expr nl

expr ::==
	func
	bitwise

func ::==
	"|" ( args "," )* "|" ( "->" type )? expr
	"|" ( args "," )* "|" ( "->" type )? "{" body "}"

args ::==
	ID ("::" type)?

bitwise ::==
	BNOT bitwise
	comparison ( (BAND | BOR | BXOR | SHL | SHR) comparison )*

comparison ::==
	NOT comparison
	arith ((GRT | GRE | LET | LEE | EQ | NEQ) arith)*

arith ::== factor ((ADD | SUB) factor)*

factor ::== unary ((MUL | DIV) unary)*

unary ::==
	primary
	(ADD | SUB) unary

primary ::==
	"(" expr ")"
	INTEGER
	REAL
	STRING
	CHAR

type ::==
	"i32"
	"u32"
	"char"
	"f32"
	"f64"
	"bool"
